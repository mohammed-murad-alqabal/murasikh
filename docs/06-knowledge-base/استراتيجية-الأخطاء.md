# 🚨 استراتيجية معالجة الأخطاء (Error Handling Strategy)

دليل شامل لكيفية معالجة الأخطاء في نظام مُرَسِّخ.

---

## 🎯 المبادئ الأساسية

### 1. مبادئ التصميم

| المبدأ | الوصف | الأولوية |
|--------|-------|----------|
| **User First** | لا تعرض أخطاء تقنية للمستخدم | 🔴 عالية |
| **Graceful Degradation** | استمر في العمل حتى مع فشل بعض المكونات | 🔴 عالية |
| **Fail Fast** | فشل سريع مع تسجيل واضح | 🟡 متوسطة |
| **Retry Logic** | إعادة المحاولة عند الأخطاء العابرة | 🟡 متوسطة |
| **Circuit Breaker** | منع الطلبات المتكررة للخدمات المعلقة | 🔴 عالية |

### 2. مستويات الخطأ

```
┌────────────────────────────────────────────────────────┐
│                 Error Severity Levels                  │
├────────────────────────────────────────────────────────┤
│                                                        │
│  🔴 CRITICAL  │ النظام لا يعمل                         │
│               │ مثال: Database Down                    │
│               │ إجراء: Alert + Fallback               │
│                                                        │
│  🟠 HIGH      │ ميزة أساسية معطلة                      │
│               │ مثال: Gemini API Unavailable          │
│               │ إجراء: Fallback + Log                 │
│                                                        │
│  🟡 MEDIUM    │ ميزة ثانوية معطلة                      │
│               │ مثال: Cache Miss                      │
│               │ إجراء: Continue + Log                 │
│                                                        │
│  🟢 LOW       │ تحذير أو تنبيه                         │
│               │ مثال: Input Validation Failed         │
│               │ إجراء: Return Error to User           │
│                                                        │
└────────────────────────────────────────────────────────┘
```

## 🔄 استراتيجيات Fallback

### 1. هرم Fallback

```
┌─────────────────────────────────────────────────────────┐
│                  Fallback Hierarchy                     │
├─────────────────────────────────────────────────────────┤
│                                                         │
│  Level 1: Normal Flow                                   │
│  ├── Gemini API → Emotion Analysis                      │
│  ├── ChromaDB → Semantic Search                         │
│  ├── PostgreSQL → Store Interaction                     │
│  └── Redis → Cache Result                               │
│                                                         │
│  Level 2: Primary Fallback                              │
│  ├── Gemini Failed → Keyword-based Analysis             │
│  ├── ChromaDB Failed → PostgreSQL Text Search           │
│  ├── PostgreSQL Failed → Memory Storage                 │
│  └── Redis Failed → No Cache (slower)                   │
│                                                         │
│  Level 3: Secondary Fallback                            │
│  ├── All AI Failed → Generic Recommendations            │
│  ├── All DB Failed → Pre-defined Content                │
│  └── System Down → Error Page with Retry                │
│                                                         │
└─────────────────────────────────────────────────────────┘
```

### 2. كود Fallback

```python
# app/services/ai/emotion_analyzer.py

class EmotionAnalyzer:
    def __init__(self):
        self.gemini_client = GeminiClient()
        self.keyword_analyzer = KeywordAnalyzer()
        
    async def analyze(self, text: str) -> dict:
        """
        تحليل المشاعر مع Fallback
        
        الأولوية:
        1. Gemini API (الأفضل)
        2. Keyword-based Analysis (Fallback)
        3. Default Response (Final Fallback)
        """
        try:
            # Level 1: Gemini API
            result = await self.gemini_client.analyze(text)
            return self._validate_result(result)
            
        except GeminiAPIError as e:
            logger.warning(f"Gemini API failed: {e}")
            
            try:
                # Level 2: Keyword-based Fallback
                result = await self.keyword_analyzer.analyze(text)
                result["fallback_used"] = "keyword_analysis"
                return result
                
            except Exception as e2:
                logger.error(f"Keyword analysis also failed: {e2}")
                
                # Level 3: Default Response
                return {
                    "emotion": "طبيعي",
                    "confidence": 0.5,
                    "fallback_used": "default",
                    "warning": "تم استخدام التحليل الافتراضي"
                }
                
    def _validate_result(self, result: dict) -> dict:
        """التحقق من صحة النتيجة"""
        valid_emotions = ['غضب', 'حزن', 'قلق', 'فرح', 'طبيعي', 'سفر', 'دراسة', 'مرض', 'شكر']
        
        if result.get("emotion") not in valid_emotions:
            result["emotion"] = "طبيعي"
            
        confidence = result.get("confidence", 0.5)
        if not 0 <= confidence <= 1:
            result["confidence"] = 0.5
            
        return result
```

### 3. Keyword-based Fallback

```python
# app/services/ai/keyword_analyzer.py

class KeywordAnalyzer:
    """
    محلل المشاعر بالكلمات المفتاحية
    Fallback عند فشل Gemini API
    """
    
    EMOTION_KEYWORDS = {
        'غضب': ['غضب', 'عصبية', 'انفعال', 'حنق', 'غيظ', 'زعل', 'غاضب'],
        'حزن': ['حزن', 'أسى', 'وجع', 'ألم', 'هم', 'كآبة', 'حزين'],
        'قلق': ['قلق', 'خوف', 'توتر', 'هم', 'خشية', 'خائف', 'متحمس'],
        'فرح': ['فرح', 'سرور', 'سعادة', 'بهجة', 'سعيد', 'مبسوط'],
        'سفر': ['سفر', 'رحلة', 'سير', 'انتقال', 'مسافر'],
        'دراسة': ['علم', 'تعلم', 'قراءة', 'دراسة', 'امتحان', 'اختبار'],
        'مرض': ['مرض', 'ألم', 'وعكة', 'شفاء', 'تعب', 'مريض'],
        'شكر': ['شكر', 'حمد', 'نعمة', 'فضل', 'امتنان']
    }
    
    async def analyze(self, text: str) -> dict:
        """تحليل النص بناءً على الكلمات المفتاحية"""
        text_lower = text.lower()
        
        # عد الكلمات المفتاحية لكل حالة
        emotion_scores = {}
        for emotion, keywords in self.EMOTION_KEYWORDS.items():
            score = sum(1 for kw in keywords if kw in text_lower)
            if score > 0:
                emotion_scores[emotion] = score
        
        if emotion_scores:
            # اختيار الحالة بأعلى درجة
            top_emotion = max(emotion_scores, key=emotion_scores.get)
            total_matches = sum(emotion_scores.values())
            
            # حساب الثقة بناءً على عدد التطابقات
            confidence = min(0.7, total_matches * 0.15 + 0.3)
            
            return {
                "emotion": top_emotion,
                "confidence": confidence,
                "method": "keyword_analysis"
            }
        
        # إذا لم يتم العثور على كلمات مفتاحية
        return {
            "emotion": "طبيعي",
            "confidence": 0.3,
            "method": "keyword_analysis"
        }
```

## 🛡️ Circuit Breaker Pattern

### 1. التكوين

```python
# app/core/circuit_breaker.py

from enum import Enum
from datetime import datetime, timedelta
import asyncio

class CircuitState(Enum):
    CLOSED = "closed"      # يعمل بشكل طبيعي
    OPEN = "open"          # مفتوح (يرفض الطلبات)
    HALF_OPEN = "half_open"  # نصف مفتوح (يسمح ببعض الطلبات)

class CircuitBreaker:
    """
    Circuit Breaker لحماية الخدمات الخارجية
    
    القواعد:
    - إذا فشلت 5 طلبات متتالية → OPEN
    - يبقى OPEN لمدة 30 ثانية
    - بعد 30 ثانية → HALF_OPEN
    - إذا نجح الطلب في HALF_OPEN → CLOSED
    - إذا فشل في HALF_OPEN → OPEN
    """
    
    def __init__(
        self,
        failure_threshold: int = 5,
        recovery_timeout: int = 30,
        half_open_max_calls: int = 1
    ):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.half_open_max_calls = half_open_max_calls
        
        self.failure_count = 0
        self.last_failure_time = None
        self.state = CircuitState.CLOSED
        self.half_open_calls = 0
        
    async def call(self, func, *args, **kwargs):
        """تنفيذ دالة مع حماية Circuit Breaker"""
        
        if self.state == CircuitState.OPEN:
            # تحقق إذا كان وقت الاسترداد
            if self._should_attempt_reset():
                self.state = CircuitState.HALF_OPEN
                self.half_open_calls = 0
            else:
                raise CircuitBreakerOpenError(
                    "Service temporarily unavailable (circuit breaker open)"
                )
        
        try:
            result = await func(*args, **kwargs)
            self._on_success()
            return result
            
        except Exception as e:
            self._on_failure()
            raise e
            
    def _should_attempt_reset(self) -> bool:
        """هل يجب محاولة إعادة التعيين؟"""
        if self.last_failure_time is None:
            return False
        return datetime.now() - self.last_failure_time > timedelta(seconds=self.recovery_timeout)
        
    def _on_success(self):
        """عند النجاح"""
        self.failure_count = 0
        self.state = CircuitState.CLOSED
        
    def _on_failure(self):
        """عند الفشل"""
        self.failure_count += 1
        self.last_failure_time = datetime.now()
        
        if self.state == CircuitState.HALF_OPEN:
            self.state = CircuitState.OPEN
            
        elif self.failure_count >= self.failure_threshold:
            self.state = CircuitState.OPEN
```

## 🔄 Retry Logic

### 1. Exponential Backoff

```python
# app/utils/retry.py

import asyncio
import random
from typing import Callable, Type, Tuple

async def retry_with_backoff(
    func: Callable,
    max_retries: int = 3,
    initial_delay: float = 1.0,
    max_delay: float = 30.0,
    exponential_base: float = 2.0,
    jitter: bool = True,
    retryable_exceptions: Tuple[Type[Exception], ...] = (Exception,)
):
    """
    إعادة المحاولة مع تأخير أسي
    
    Args:
        func: الدالة المراد تنفيذها
        max_retries: أقصى عدد من المحاولات
        initial_delay: التأخير الأولي (ثواني)
        max_delay: أقصى تأخير (ثواني)
        exponential_base: الأساس الأسي
        jitter: إضافة عشوائية للتأخير
        retryable_exceptions: الاستثناءات التي تستدعي إعادة المحاولة
        
    Returns:
        نتيجة الدالة
        
    Raises:
        آخر استثناء حدث بعد استنفاد المحاولات
    """
    retry = 0
    last_exception = None
    
    while retry <= max_retries:
        try:
            return await func()
            
        except retryable_exceptions as e:
            last_exception = e
            retry += 1
            
            if retry > max_retries:
                logger.error(f"Max retries ({max_retries}) exceeded")
                raise e
                
            # حساب التأخير
            delay = min(initial_delay * (exponential_base ** (retry - 1)), max_delay)
            
            # إضافة jitter
            if jitter:
                delay = delay * (0.5 + random.random())
                
            logger.warning(
                f"Retry {retry}/{max_retries} after {delay:.2f}s due to: {e}"
            )
            
            await asyncio.sleep(delay)
            
    raise last_exception
```

```python
# app/core/logging.py

import logging
import json
from datetime import datetime
from typing import Any, Dict

class StructuredLogger:
    """تسجيل منظم للأخطاء"""
    
    def __init__(self, name: str):
        self.logger = logging.getLogger(name)
        
    def error(
        self,
        message: str,
        error: Exception = None,
        context: Dict[str, Any] = None,
        user_id: int = None,
        request_id: str = None
    ):
        """تسجيل خطأ مع السياق"""
        log_data = {
            "timestamp": datetime.utcnow().isoformat(),
            "level": "ERROR",
            "message": message,
            "error_type": type(error).__name__ if error else None,
            "error_message": str(error) if error else None,
            "context": context or {},
            "user_id": user_id,
            "request_id": request_id
        }
        
        self.logger.error(json.dumps(log_data, ensure_ascii=False))
        
        # إرسال إلى Sentry
        if error:
            self._send_to_sentry(error, log_data)
            
    def _send_to_sentry(self, error: Exception, log_data: dict):
        """إرسال الخطأ إلى Sentry"""
        try:
            import sentry_sdk
            sentry_sdk.capture_exception(
                error,
                extras=log_data
            )
        except ImportError:
            pass

# استخدام
logger = StructuredLogger(__name__)

try:
    result = await some_function()
except Exception as e:
    logger.error(
        message="Failed to process request",
        error=e,
        context={"text": user_text, "emotion": emotion},
        user_id=user_id,
        request_id=request_id
    )
```
