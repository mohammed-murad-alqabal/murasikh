## 🎯 اختبارات الأداء (Performance Tests)

### 1. Load Testing Script (k6)

```javascript
// tests/load/recommendation_load.js

import http from 'k6/http';
import { check, sleep } from 'k6';
import { Rate } from 'k6/metrics';

const errorRate = new Rate('errors');

export let options = {
  stages: [
    { duration: '2m', target: 10 },   // Ramp up to 10 users
    { duration: '5m', target: 50 },   // Stay at 50 users
    { duration: '2m', target: 100 },  // Ramp up to 100 users
    { duration: '5m', target: 100 },  // Stay at 100 users
    { duration: '2m', target: 0 },    // Ramp down
  ],
  thresholds: {
    http_req_duration: ['p(50)<1000', 'p(95)<2000', 'p(99)<3000'],
    errors: ['rate<0.05'], // Error rate < 5%
  },
};

const BASE_URL = __ENV.BASE_URL || 'http://localhost:8000';

const SAMPLE_TEXTS = [
  'أشعر بالقلق من المستقبل',
  'أنا غاضب جداً من هذا الموقف',
  'أحس بالحزن الشديد',
  'أشعر بالفرح والسعادة اليوم',
];

export default function () {
  const text = SAMPLE_TEXTS[Math.floor(Math.random() * SAMPLE_TEXTS.length)];
  
  const payload = JSON.stringify({
    text: text,
  });

  const params = {
    headers: {
      'Content-Type': 'application/json',
    },
  };

  const res = http.post(`${BASE_URL}/api/v1/recommend`, payload, params);

  const success = check(res, {
    'status is 200': (r) => r.status === 200,
    'response time < 2000ms': (r) => r.timings.duration < 2000,
    'has recommendation': (r) => {
      const body = JSON.parse(r.body);
      return body.success && body.data.recommendation;
    },
  });

  errorRate.add(!success);

  sleep(1);
}
```

### 2. Stress Testing Script (Locust)

```python
# tests/load/locustfile.py

from locust import HttpUser, task, between
import random

class RecommendationUser(HttpUser):
    wait_time = between(1, 3)
    
    SAMPLE_TEXTS = [
        'أشعر بالقلق من المستقبل',
        'أنا غاضب جداً من هذا الموقف',
        'أحس بالحزن الشديد',
        'أشعر بالفرح والسعادة اليوم',
        'أريد أن أشكر الله على النعم',
    ]
    
    @task(10)
    def get_recommendation(self):
        """طلب توصية عشوائي"""
        text = random.choice(self.SAMPLE_TEXTS)
        
        self.client.post(
            '/api/v1/recommend',
            json={'text': text},
            name='/recommend'
        )
    
    @task(2)
    def send_feedback(self):
        """إرسال تغذية راجعة"""
        interaction_id = random.randint(1000, 10000)
        
        self.client.post(
            '/api/v1/feedback',
            json={
                'interaction_id': interaction_id,
                'rating': random.choice(['positive', 'negative', 'neutral'])
            },
            name='/feedback'
        )
    
    @task(1)
    def health_check(self):
        """فحص الصحة"""
        self.client.get('/api/v1/health', name='/health')
```
